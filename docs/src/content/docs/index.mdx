import { Card, CardGrid, LinkCard, Tabs, TabItem } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="Small and cheap" icon="rocket">
    A snapshot stores names and object ids only, so taking one is fast and the JSON is tiny.
  </Card>
  <Card title="Safe restores" icon="approve-check">
    `Restore` checks everything first. Missing objects or uncommitted changes stop it before anything is touched.
  </Card>
  <Card title="Made for tests" icon="puzzle">
    Wrap a test in `GitRepository.Preserve(path)` and every change to the repository is undone on dispose.
  </Card>
  <Card title="Library and CLI" icon="seti:shell">
    Use the `GitSnapshotter` package from .NET, or the `gitsnapshotter` tool from a shell script.
  </Card>
</CardGrid>

## At a glance

<Tabs syncKey="interface">
  <TabItem label="C#">
    ```csharp
    using GitSnapshotter;

    var snapshot = GitRepository.GetSnapshot(path);

    // rebase, delete branches, move tags, add remotes...

    GitRepository.Restore(path, snapshot);
    ```
  </TabItem>
  <TabItem label="Command line">
    ```sh
    gitsnapshotter snapshot . -o before.json

    # rebase, delete branches, move tags, add remotes...

    gitsnapshotter restore . before.json
    ```
  </TabItem>
</Tabs>
