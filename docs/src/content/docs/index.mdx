import { Card, CardGrid } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="Small and cheap" icon="rocket">
    A snapshot stores names and object ids only, so taking one is fast and the JSON is tiny.
  </Card>
  <Card title="Safe restores" icon="approve-check">
    `Restore` checks everything first. Missing objects or uncommitted changes stop it before anything is touched.
  </Card>
  <Card title="Made for tests" icon="puzzle">
    Wrap a test in `GitRepository.Preserve(path)` and every change to the repository is undone on dispose.
  </Card>
  <Card title="Library and CLI" icon="seti:shell">
    Use the `GitSnapshotter` NuGet package from .NET, or the `GitSnapshotter.Console` tool from a shell.
  </Card>
</CardGrid>
